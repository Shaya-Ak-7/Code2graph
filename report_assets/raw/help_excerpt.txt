$ code2graph --help      # command summary (options omitted)
  extract <path>           headless full extraction of a repository
  update <path>            re-extract code files and update the graph (no LLM)
  watch <path>             watch a folder and rebuild the graph on code changes
  cluster-only <path>      rerun clustering on an existing graph.json
  label <path>             (re)name communities with the configured backend
  query "<question>"       BFS traversal of graph.json for a question
  path "A" "B"             shortest path between two nodes in graph.json
  explain "X"              plain-language explanation of a node
  affected "X"             reverse traversal to find nodes impacted by X
  god-nodes                list the most connected nodes (architectural hubs)
  export html              emit interactive graph.html
  export callflow-html     emit Mermaid-based architecture / call-flow HTML
  export obsidian          emit Obsidian vault notes + canvas
  export wiki              emit wiki markdown articles
  export svg               emit graph.svg
  export graphml           emit GraphML
  export neo4j             emit Cypher or push to Neo4j
  export falkordb          emit Cypher or push to FalkorDB
  merge-graphs <g1> <g2>   merge two or more graph.json files into one
  global add <graph>       add or update a project graph in the global graph
  install [--platform]     copy the skill to a coding-agent config folder
  hook install             install post-commit / post-checkout git hooks
  prs                      PR dashboard: CI state, review status, worktrees
  benchmark [graph.json]   measure token reduction vs full-corpus reading
  diagnose multigraph      report same-endpoint edge collapse risk
